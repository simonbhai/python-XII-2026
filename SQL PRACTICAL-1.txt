Q3. Create the Student table.
CREATE TABLE Student (
    ADMNO VARCHAR(10) PRIMARY KEY,
    NAME VARCHAR(20) NOT NULL,
    STREAM VARCHAR(15),
    DOB DATE,
    GENDER CHAR(1),
    OPTION1 VARCHAR(5),
    MARKS INT
);

Q7. Show details of Commerce and Humanities students.

Q9. Display female Commerce students.

Q10. Display ADMNO, NAME and MARKS of male students.

Q11. Display ADMNO, NAME, STREAM and the overall average marks with heading Average Marks.
SELECT ADMNO, NAME, STREAM,
       (SELECT AVG(MARKS) FROM Student) AS `Average Marks`
FROM Student;

Q12. Display students with marks from 415 to 418.
SELECT * FROM Student WHERE MARKS BETWEEN 415 AND 418;

Q13. Display names that start with A and end with A.

Q15. Display streams without repetition.

Q16. Display names in descending order of marks.

Q17. Display students born before 1 December 2004.

Q18. Delete the record of a student who scored 475.

Q19. Count the students.
SELECT COUNT(*) AS Total_Students FROM Student;

Q20. Count the records/games in Sports.
SELECT COUNT(*) AS Total_Games FROM Sports;

Q21. Show minimum marks in Science.
SELECT MIN(MARKS) AS Minimum_Marks FROM Student WHERE STREAM = 'Science';

Q22. Show maximum marks of male students.
SELECT MAX(MARKS) AS Maximum_Marks FROM Student WHERE GENDER = 'M';

Q23. Find the sum of marks greater than 470.
SELECT SUM(MARKS) AS Total_Marks FROM Student WHERE MARKS > 470;

Q24. Display names and games of students whose optional subject is IP.
SELECT Student.NAME, Sports.GAME
FROM Student JOIN Sports ON Student.ADMNO = Sports.ADMNO
WHERE Student.OPTION1 = 'IP';

Q25. Display Science students' names and coach names.
SELECT Student.NAME, Sports.COACHNAME
FROM Student JOIN Sports ON Student.ADMNO = Sports.ADMNO
WHERE Student.STREAM = 'Science';
